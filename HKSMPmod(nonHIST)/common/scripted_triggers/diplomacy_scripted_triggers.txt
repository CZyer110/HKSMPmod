#	Triggers to control if diplomatic actions are enabled:
#		Format:
#
#		<diplomatic action tag>_enabled_trigger = {
#			<trigger contents>
#		}
#
#		ROOT is the country initiating the diplomatic action.
#		FROM is the recipient or target of the diplomatic action.
#
#		Note:  For cleaner tooltips, it is recommended that you use
#			if triggers within the enable triggers, and a single
#			custom_trigger_tooltip trigger within each if trigger.
#
#		Example: (Germany is prevented from declaring war on the Soviet Union.)
#
#		DIPLOMACY_WAR_ENABLE_TRIGGER = {
#			if = {
#				limit = {
#					tag = GER
#				}
#				custom_trigger_tooltip = {
#					tooltip = PREVENT_GER_WAR_ON_SOV
#					FROM = {
#						NOT = {
#							tag = SOV
#						}
#					}
#				}
#			}
#		}

DIPLOMACY_WAR_ENABLE_TRIGGER = {
        # ПРОВЕРКА 1: Против JAP and CHI можно только после 1940
		if = {
			limit = {
				NOT = {has_country_flag = panay_no_sorry}
				OR = {
					FROM = {
						has_war_with = JAP
						OR = {
							tag = CHI
							tag = PRC
							tag = SND
							tag = HBC
							tag = SHX
							tag = GDC
							tag = GXC
							tag = YUN
							tag = SIC
							tag = XIC
							tag = XSM
							tag = GSM
							tag = NXM
							tag = KHM
							tag = SIK
						}
					}
					FROM = {
						has_war_with = CHI
						OR = {
							tag = JAP
							tag = MAN
							tag = MEN
						}
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = RULE_BAN_INTERVENTIONS_IN_JAP_CHI
				date > 1940.1.1
			}
		}
		# ПРОВЕРКА 1.2: Против SOV или RUS можно только после 1940
        else_if = {
                limit = {
						is_ai = no
                        FROM = {
                                OR = {
                                        tag = SOV
                                        tag = RUS
										is_puppet_of = SOV
										is_puppet_of = RUS
										is_guaranteed_by = SOV
										is_guaranteed_by = RUS
                                }
                        }
                }
				custom_trigger_tooltip = {
					tooltip = RULE_BAN_WAR_WITH_SOV
					date > 1940.1.1
				}
        }
		# ПРОВЕРКА 1.3: Против JAP можно только после 1940
		else_if = {
                limit = {
						is_ai = no
                        FROM = {
                                OR = {
                                        tag = JAP
										is_puppet_of = JAP
										is_guaranteed_by = JAP
										tag = BRA
										is_puppet_of = BRA
										is_guaranteed_by = BRA
                                }
                        }
                }
				custom_trigger_tooltip = {
					tooltip = RULE_BAN_WAR_WITH_JAP_OR_BRA
					date > 1937.1.1
				}
        }
        # ПРОВЕРКА 2: Против игрока Мажора/Супермажора — после 1939
        else_if = {
                limit = {
					is_ai = no
					OR = {
                        FROM = {
                                is_ai = no
                                OR = {
                                        has_country_flag = custom_status_supermajor
                                        has_country_flag = custom_status_major
                                }
						}
						FROM = {
								is_subject = yes
								overlord = {
									OR = { has_country_flag = custom_status_supermajor has_country_flag = custom_status_major}
									is_ai = no
								}
						}
						any_country = {
							OR = { has_country_flag = custom_status_supermajor has_country_flag = custom_status_major}
							is_ai = no
							has_guaranteed = FROM
						}
					}
                }
				custom_trigger_tooltip = {
					tooltip = RULE_BAN_WAR_WITH_MAJOR_PLAYERS
					date > 1939.1.1
				}
        }
        # ПРОВЕРКА 3: Против ИИ Мажора/Супермажора — после 1938
        else_if = {
                limit = {
					is_ai = no
					OR = {
                        FROM = {
                                is_ai = yes
                                OR = {
                                        has_country_flag = custom_status_supermajor
                                        has_country_flag = custom_status_major
                                }
						}
						FROM = {
								is_subject = yes
								overlord = {
									OR = { has_country_flag = custom_status_supermajor has_country_flag = custom_status_major}
									is_ai = yes
								}
						}
						any_country = {
							OR = { has_country_flag = custom_status_supermajor has_country_flag = custom_status_major}
							is_ai = yes
							has_guaranteed = FROM
						}
					}
                }
				custom_trigger_tooltip = {
					tooltip = RULE_BAN_WAR_WITH_MAJOR_AI
					date > 1938.1.1
				}
        }
		# ПРОВЕРКА 6: Миноры в Азии/Америках против таких же миноров (Всегда можно)
		else_if = {
                limit = {
						is_ai = no
                        capital_scope = {
							is_on_continent = asia
                        }
                        FROM = {
                                has_country_flag = custom_status_minor
                                capital_scope = {
									is_on_continent = asia
                                }
                        }
                }
                always = yes
        }
		else_if = {
                limit = {
						is_ai = no
                        has_country_flag = custom_status_minor
                        capital_scope = {
							is_on_continent = south_america
                        }
                        FROM = {
                                has_country_flag = custom_status_minor
                                capital_scope = {
									is_on_continent = south_america
                                }
                        }
                }
                always = yes
        }
		# ПРОВЕРКА 4: Против игрока Минора — после 1938
        else_if = {
                limit = {
					is_ai = no
					OR = {
                        FROM = {
                                is_ai = no
                                has_country_flag = custom_status_minor
						}
						FROM = {
								is_subject = yes
								overlord = {
									has_country_flag = custom_status_minor
									is_ai = no
								}
						}
						any_country = {
							has_country_flag = custom_status_minor
							is_ai = no
							has_guaranteed = FROM
						}
					}
                }
				custom_trigger_tooltip = {
					tooltip = RULE_BAN_WAR_WITH_MINOR_PLAYERS
					date > 1938.1.1
				}
        }
		else_if = {
                limit = {
					tag = ENG
					has_civil_war = yes
                    FROM = {NOT = {tag = MAL}}
                }
				custom_trigger_tooltip = {
					tooltip = RULE_BAN_WAR_WITH_DOMINIONS_IN_CIVIL_WAR
					always = no
				}
        }
		else_if = {
                limit = {
					tag = ENG
					has_civil_war = yes
                    FROM = {tag = MAL}
                }
				always = yes
        }
        # ПРОВЕРКА 5: Против ИИ Минора — после 1937
		else_if = {
			limit = {
				OR = {
					FROM = {
						is_ai = yes
						has_country_flag = custom_status_minor
					}
					FROM = {
						is_subject = yes
						overlord = {
							has_country_flag = custom_status_minor
							is_ai = yes
						}
					}
					any_country = {
						has_country_flag = custom_status_minor
						is_ai = yes
						has_guaranteed = FROM
					}
				}
			}
			custom_trigger_tooltip = {
				tooltip = RULE_BAN_WAR_WITH_MINOR_AI
				date > 1937.1.1
			}
        }
        else = {
                always = yes
        }
}

# This specifically overrides the rule check as to whether or not you can guarantee other ideologies
DIPLOMACY_GUARANTEE_ENABLE_TRIGGER_OVERRIDES_GAME_TRIGGER = {
	NOT = {
		has_game_rule = {
			rule = allow_guarantees
			option = LIMITED
		}
	}
}

DIPLOMACY_GUARANTEE_ENABLE_TRIGGER = {
	if = {
		limit = {
			any_country = {
				OR = {
					is_justifying_wargoal_against = FROM
					has_wargoal_against = FROM
				}
			}
		}
		custom_trigger_tooltip = {
			tooltip = RULE_BAN_GUARANTEE_WHILE_JUSTIFYING_WARGOAL
			always = no
		}
	}

	if = {
		limit = {
			has_game_rule = {
				rule = allow_guarantees
				option = ALWAYS_FREE
			}
		}
		always = yes
	}
	if = {
		limit = {
			has_game_rule = {
				rule = allow_guarantees
				option = LIMITED
			}
		}
	}
	if = {
		limit = {
			has_game_rule = {
				rule = allow_guarantees
				option = SAME_IDEOLOGY
			}
		}
		custom_trigger_tooltip = {
			tooltip = RULE_ALLOW_GUARANTEES_SAME_IDEOLOGY_TOOLTIP
			has_government = FROM
		}
	}
	if = {
		limit = {
			has_game_rule = {
				rule = allow_guarantees
				option = BLOCKED
			}
		}
		custom_trigger_tooltip = {
			tooltip = RULE_ALLOW_GUARANTEES_BLOCKED_TOOLTIP
			always = no
		}
	}
}
